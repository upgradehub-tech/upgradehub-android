import React, { useState, useEffect, useRef, useCallback } from 'react';
import {
  TestPaper,
  QuestionResponse,
  QuestionStatus,
  TestEvaluationResult
} from '../types/cbt';
import { evaluateCBTSubmission } from '../engine/scoringEngine';

interface CBTProps {
  paper: TestPaper;
  onFinishTest: (result: TestEvaluationResult) => void;
}

export const CBTTestArena: React.FC<CBTProps> = ({ paper, onFinishTest }) => {
  const [activeSectionId, setActiveSectionId] = useState<string>(paper.sections[0].id);
  const [activeQuestionId, setActiveQuestionId] = useState<string>(paper.sections[0].questionIds[0]);
  const [responses, setResponses] = useState<Record<string, QuestionResponse>>(() => {
    const initial: Record<string, QuestionResponse> = {};
    paper.sections.forEach(sec => {
      sec.questionIds.forEach((qId, idx) => {
        initial[qId] = {
          questionId: qId,
          status: idx === 0 && sec.id === paper.sections[0].id ? 'NOT_ANSWERED' : 'NOT_VISITED',
          timeSpentSeconds: 0
        };
      });
    });
    return initial;
  });

  const [timeRemaining, setTimeRemaining] = useState<number>(paper.durationSeconds);
  const [isSubmitModalOpen, setIsSubmitModalOpen] = useState<boolean>(false);

  // Time Tracking Refs for drift resistance
  const lastTickRef = useRef<number>(Date.now());
  const activeQuestionStartTimeRef = useRef<number>(Date.now());

  const currentSection = paper.sections.find(s => s.id === activeSectionId)!;
  const currentQuestion = paper.questions[activeQuestionId];
  const currentResponse = responses[activeQuestionId];

  // Final submission handler
  const finalizeSubmission = useCallback(() => {
    // Record residual time for the current question
    const now = Date.now();
    const elapsedOnCurrent = Math.floor((now - activeQuestionStartTimeRef.current) / 1000);
    const updatedResponses = {
      ...responses,
      [activeQuestionId]: {
        ...responses[activeQuestionId],
        timeSpentSeconds: responses[activeQuestionId].timeSpentSeconds + elapsedOnCurrent
      }
    };
    const totalTimeTaken = paper.durationSeconds - timeRemaining;
    const result = evaluateCBTSubmission(paper, updatedResponses, totalTimeTaken);
    onFinishTest(result);
  }, [paper, responses, activeQuestionId, timeRemaining, onFinishTest]);

  // Real Countdown Timer with background tab drift handling
  useEffect(() => {
    const interval = setInterval(() => {
      const now = Date.now();
      const deltaSec = Math.floor((now - lastTickRef.current) / 1000);

      if (deltaSec >= 1) {
        lastTickRef.current = now;
        setTimeRemaining(prev => {
          if (prev <= 1) {
            clearInterval(interval);
            finalizeSubmission();
            return 0;
          }
          return prev - deltaSec;
        });
      }
    }, 250);

    return () => clearInterval(interval);
  }, [finalizeSubmission]);

  // Flush time spent on question change
  const navigateToQuestion = (targetQId: string) => {
    if (targetQId === activeQuestionId) return;

    const now = Date.now();
    const elapsedSeconds = Math.max(1, Math.floor((now - activeQuestionStartTimeRef.current) / 1000));
    activeQuestionStartTimeRef.current = now;

    setResponses(prev => {
      const current = prev[activeQuestionId];
      const target = prev[targetQId];

      return {
        ...prev,
        [activeQuestionId]: {
          ...current,
          timeSpentSeconds: current.timeSpentSeconds + elapsedSeconds,
          status: current.status === 'NOT_VISITED' ? 'NOT_ANSWERED' : current.status
        },
        [targetQId]: {
          ...target,
          status: target.status === 'NOT_VISITED' ? 'NOT_ANSWERED' : target.status
        }
      };
    });

    const targetSection = paper.sections.find(s => s.questionIds.includes(targetQId));
    if (targetSection && targetSection.id !== activeSectionId) {
      setActiveSectionId(targetSection.id);
    }
    setActiveQuestionId(targetQId);
  };

  const handleOptionSelect = (optionId: string) => {
    setResponses(prev => ({
      ...prev,
      [activeQuestionId]: {
        ...prev[activeQuestionId],
        selectedOptionId: optionId
      }
    }));
  };

  const handleNumericalChange = (val: string) => {
    // Only allow sanitized decimal input
    if (/^-?\d*\.?\d*$/.test(val)) {
      setResponses(prev => ({
        ...prev,
        [activeQuestionId]: {
          ...prev[activeQuestionId],
          numericalValue: val
        }
      }));
    }
  };

  const handleSaveAndNext = () => {
    const hasValue =
      currentQuestion.type === 'MCQ'
        ? !!currentResponse.selectedOptionId
        : currentResponse.numericalValue !== undefined && currentResponse.numericalValue.trim() !== '';

    setResponses(prev => ({
      ...prev,
      [activeQuestionId]: {
        ...prev[activeQuestionId],
        status: hasValue ? 'ANSWERED' : 'NOT_ANSWERED'
      }
    }));

    advanceNext();
  };

  const handleMarkForReviewAndNext = () => {
    const hasValue =
      currentQuestion.type === 'MCQ'
        ? !!currentResponse.selectedOptionId
        : currentResponse.numericalValue !== undefined && currentResponse.numericalValue.trim() !== '';

    setResponses(prev => ({
      ...prev,
      [activeQuestionId]: {
        ...prev[activeQuestionId],
        status: hasValue ? 'ANSWERED_AND_MARKED' : 'MARKED_FOR_REVIEW'
      }
    }));

    advanceNext();
  };

  const handleClearResponse = () => {
    setResponses(prev => ({
      ...prev,
      [activeQuestionId]: {
        ...prev[activeQuestionId],
        selectedOptionId: undefined,
        numericalValue: '',
        status: 'NOT_ANSWERED'
      }
    }));
  };

  const advanceNext = () => {
    const currList = currentSection.questionIds;
    const currentIndex = currList.indexOf(activeQuestionId);

    if (currentIndex < currList.length - 1) {
      navigateToQuestion(currList[currentIndex + 1]);
    } else {
      // Find next section
      const secIdx = paper.sections.findIndex(s => s.id === activeSectionId);
      if (secIdx < paper.sections.length - 1) {
        const nextSec = paper.sections[secIdx + 1];
        setActiveSectionId(nextSec.id);
        navigateToQuestion(nextSec.questionIds[0]);
      }
    }
  };

  // Format Timer String
  const formatTime = (secs: number) => {
    const h = Math.floor(secs / 3600);
    const m = Math.floor((secs % 3600) / 60);
    const s = secs % 60;
    return `${String(h).padStart(2, '0')}:${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`;
  };

  // Status Styling Generator for NTA Palette
  const getStatusColor = (status: QuestionStatus) => {
    switch (status) {
      case 'ANSWERED':
        return 'bg-emerald-600 text-white';
      case 'NOT_ANSWERED':
        return 'bg-amber-600 text-white';
      case 'MARKED_FOR_REVIEW':
        return 'bg-purple-600 text-white';
      case 'ANSWERED_AND_MARKED':
        return 'bg-purple-800 border-2 border-emerald-400 text-white';
      case 'NOT_VISITED':
      default:
        return 'bg-slate-200 text-slate-800 hover:bg-slate-300';
    }
  };

  // Stats Breakdown for Confirmation Modal
  const statusCounts = Object.values(responses).reduce(
    (acc, r) => {
      acc[r.status] = (acc[r.status] || 0) + 1;
      return acc;
    },
    {} as Record<QuestionStatus, number>
  );

  return (
    <div className="flex flex-col h-screen w-screen bg-slate-50 font-sans select-none overflow-hidden">
      {/* Top Header */}
      <header className="h-16 bg-slate-900 text-white px-6 flex items-center justify-between border-b border-slate-700">
        <div>
          <h1 className="text-lg font-bold tracking-wide">{paper.title}</h1>
          <span className="text-xs text-slate-400">Section: {currentSection.name}</span>
        </div>
        <div className="flex items-center space-x-6">
          <div className="bg-slate-800 px-4 py-2 rounded-lg border border-slate-600 flex items-center space-x-2">
            <span className="text-xs uppercase text-slate-400 tracking-wider">Time Left:</span>
            <span className={`font-mono text-xl font-bold ${timeRemaining < 300 ? 'text-rose-500 animate-pulse' : 'text-emerald-400'}`}>
              {formatTime(timeRemaining)}
            </span>
          </div>
          <button
            onClick={() => setIsSubmitModalOpen(true)}
            className="bg-rose-600 hover:bg-rose-700 text-white text-sm font-semibold px-5 py-2 rounded shadow transition"
          >
            Submit Paper
          </button>
        </div>
      </header>

      {/* Main Workspace */}
      <div className="flex flex-1 overflow-hidden">
        {/* Left Column: Test Area */}
        <div className="flex-1 flex flex-col justify-between p-6 overflow-y-auto">
          {/* Section Selection Bar */}
          <div className="flex items-center space-x-2 border-b border-slate-200 pb-3 mb-4">
            {paper.sections.map(s => (
              <button
                key={s.id}
                onClick={() => {
                  setActiveSectionId(s.id);
                  navigateToQuestion(s.questionIds[0]);
                }}
                className={`px-4 py-2 text-sm font-medium rounded-t border-b-2 transition ${
                  activeSectionId === s.id
                    ? 'border-indigo-600 text-indigo-600 bg-white shadow-sm'
                    : 'border-transparent text-slate-500 hover:text-slate-800'
                }`}
              >
                {s.name}
              </button>
            ))}
          </div>

          {/* Question Viewer */}
          <div className="bg-white p-6 rounded-lg shadow-sm border border-slate-200 flex-1 overflow-y-auto">
            <div className="flex justify-between items-center mb-4 pb-2 border-b border-slate-100">
              <span className="font-bold text-slate-700">
                Question {currentSection.questionIds.indexOf(activeQuestionId) + 1}
              </span>
              <div className="text-xs space-x-3 text-slate-500">
                <span>Marks: <span className="text-emerald-600 font-semibold">+{currentQuestion.markingScheme.correct}</span></span>
                <span>Negative: <span className="text-rose-600 font-semibold">-{Math.abs(currentQuestion.markingScheme.incorrect)}</span></span>
              </div>
            </div>

            <p className="text-slate-800 text-base leading-relaxed mb-6">
              {currentQuestion.prompt}
            </p>

            {/* MCQ Options Rendering */}
            {currentQuestion.type === 'MCQ' && currentQuestion.options && (
              <div className="space-y-3">
                {currentQuestion.options.map(opt => (
                  <label
                    key={opt.id}
                    onClick={() => handleOptionSelect(opt.id)}
                    className={`flex items-center p-3 rounded-md border cursor-pointer transition ${
                      currentResponse.selectedOptionId === opt.id
                        ? 'bg-indigo-50 border-indigo-600 text-indigo-900 font-medium'
                        : 'border-slate-300 hover:bg-slate-50 text-slate-700'
                    }`}
                  >
                    <input
                      type="radio"
                      name={`option_${activeQuestionId}`}
                      checked={currentResponse.selectedOptionId === opt.id}
                      onChange={() => handleOptionSelect(opt.id)}
                      className="h-4 w-4 text-indigo-600 focus:ring-indigo-500"
                    />
                    <span className="ml-3 text-sm">{opt.text}</span>
                  </label>
                ))}
              </div>
            )}

            {/* Numerical Input Area */}
            {currentQuestion.type === 'NUMERICAL' && (
              <div className="mt-4">
                <label className="block text-xs font-semibold text-slate-500 uppercase tracking-wider mb-2">
                  Enter Decimal or Integer Value:
                </label>
                <input
                  type="text"
                  value={currentResponse.numericalValue || ''}
                  onChange={e => handleNumericalChange(e.target.value)}
                  placeholder="e.g. 42.5"
                  className="w-48 text-lg font-mono px-3 py-2 border rounded-md focus:outline-none focus:ring-2 focus:ring-indigo-500"
                />
              </div>
            )}
          </div>

          {/* Action Button Row */}
          <div className="flex justify-between items-center mt-4 pt-4 border-t border-slate-200 bg-slate-50">
            <div className="space-x-3">
              <button
                onClick={handleMarkForReviewAndNext}
                className="bg-purple-600 hover:bg-purple-700 text-white text-sm font-semibold px-4 py-2 rounded shadow transition"
              >
                Mark for Review & Next
              </button>
              <button
                onClick={handleClearResponse}
                className="bg-slate-200 hover:bg-slate-300 text-slate-700 text-sm font-medium px-4 py-2 rounded transition"
              >
                Clear Response
              </button>
            </div>
            <button
              onClick={handleSaveAndNext}
              className="bg-emerald-600 hover:bg-emerald-700 text-white text-sm font-bold px-6 py-2 rounded shadow transition"
            >
              Save & Next
            </button>
          </div>
        </div>

        {/* Right Column: Palette Sidebar */}
        <div className="w-80 bg-white border-l border-slate-200 flex flex-col justify-between p-4 shadow-sm">
          <div>
            <h2 className="text-sm font-bold text-slate-700 uppercase tracking-wider mb-3">
              Question Palette
            </h2>

            {/* Status Legend */}
            <div className="grid grid-cols-2 gap-2 text-xs mb-4 p-3 bg-slate-50 border rounded text-slate-600">
              <div className="flex items-center space-x-1.5">
                <span className="w-3 h-3 rounded-full bg-emerald-600 inline-block"></span>
                <span>Answered ({statusCounts['ANSWERED'] || 0})</span>
              </div>
              <div className="flex items-center space-x-1.5">
                <span className="w-3 h-3 rounded-full bg-amber-600 inline-block"></span>
                <span>Not Ans ({statusCounts['NOT_ANSWERED'] || 0})</span>
              </div>
              <div className="flex items-center space-x-1.5">
                <span className="w-3 h-3 rounded-full bg-purple-600 inline-block"></span>
                <span>Review ({statusCounts['MARKED_FOR_REVIEW'] || 0})</span>
              </div>
              <div className="flex items-center space-x-1.5">
                <span className="w-3 h-3 rounded-full bg-slate-300 inline-block"></span>
                <span>Not Visited ({statusCounts['NOT_VISITED'] || 0})</span>
              </div>
            </div>

            {/* Palette Grid */}
            <div className="grid grid-cols-5 gap-2 max-h-96 overflow-y-auto p-1">
              {currentSection.questionIds.map((qId, index) => {
                const resp = responses[qId];
                const isActive = qId === activeQuestionId;
                return (
                  <button
                    key={qId}
                    onClick={() => navigateToQuestion(qId)}
                    className={`h-10 w-10 text-sm font-bold rounded-lg flex items-center justify-center relative transition ${
                      getStatusColor(resp.status)
                    } ${isActive ? 'ring-2 ring-indigo-500 ring-offset-2' : ''}`}
                  >
                    {index + 1}
                    {resp.status === 'ANSWERED_AND_MARKED' && (
                      <span className="absolute bottom-1 right-1 w-2 h-2 rounded-full bg-emerald-300"></span>
                    )}
                  </button>
                );
              })}
            </div>
          </div>

          <div className="border-t pt-3 text-center text-xs text-slate-400">
            Total Questions: {paper.sections.reduce((acc, s) => acc + s.questionIds.length, 0)}
          </div>
        </div>
      </div>

      {/* Submission Confirmation Modal */}
      {isSubmitModalOpen && (
        <div className="fixed inset-0 bg-slate-900 bg-opacity-60 flex items-center justify-center p-4 z-50">
          <div className="bg-white rounded-lg max-w-md w-full p-6 shadow-xl">
            <h3 className="text-lg font-bold text-slate-800 mb-2">Confirm Test Submission</h3>
            <p className="text-sm text-slate-600 mb-4">
              Are you sure you want to finish the test? Review your response summary:
            </p>

            <div className="space-y-2 text-sm bg-slate-50 p-3 rounded border mb-4">
              <div className="flex justify-between">
                <span>Answered:</span>
                <span className="font-bold text-emerald-600">{(statusCounts['ANSWERED'] || 0) + (statusCounts['ANSWERED_AND_MARKED'] || 0)}</span>
              </div>
              <div className="flex justify-between">
                <span>Marked for Review (Unanswered):</span>
                <span className="font-bold text-purple-600">{statusCounts['MARKED_FOR_REVIEW'] || 0}</span>
              </div>
              <div className="flex justify-between">
                <span>Not Answered:</span>
                <span className="font-bold text-amber-600">{statusCounts['NOT_ANSWERED'] || 0}</span>
              </div>
              <div className="flex justify-between">
                <span>Not Visited:</span>
                <span className="font-bold text-slate-400">{statusCounts['NOT_VISITED'] || 0}</span>
              </div>
            </div>

            <div className="flex justify-end space-x-3">
              <button
                onClick={() => setIsSubmitModalOpen(false)}
                className="px-4 py-2 border rounded text-slate-700 text-sm hover:bg-slate-100"
              >
                Resume Test
              </button>
              <button
                onClick={finalizeSubmission}
                className="px-4 py-2 bg-rose-600 hover:bg-rose-700 text-white rounded text-sm font-semibold"
              >
                Yes, Submit
              </button>
            </div>
          </div>
        </div>
      )}
    </div>
  );
};
